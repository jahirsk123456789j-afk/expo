import React, { useState } from 'react';
import './App.css'; // তোমার স্টাইলের জন্য

const videos = [
  'video1.mp4',
  'video2.mp4',
  'video3.mp4',
  // আরো ভিডিও যোগ করো
];

function App() {
  const [currentIndex, setCurrentIndex] = useState(0);

  const nextVideo = () => {
    setCurrentIndex((currentIndex + 1) % videos.length);
  };

  return (
    <div className="app" onClick={nextVideo}>
      <video 
        src={videos[currentIndex]} 
        controls 
        autoPlay
        style={{ width: '100%', height: '100vh' }}
      />
    </div>
  );
}

export default App;
